! ============================================================
! Cloud Router
! Device  : Cisco 2911
! Role    : External email server gateway + RIPv2 + Static route back to university
! Serial  : Se0/2/0 → Main Campus Router (20.0.0.0/30)
! Note    : Simulates the ISP/cloud side hosting the external email server
! ============================================================

hostname Cloud-Router

! -------------------------------------------------------
! Bring up all interfaces
! -------------------------------------------------------
interface GigabitEthernet0/0
 no shutdown

interface Serial0/2/0
 no shutdown

! -------------------------------------------------------
! Serial Interface IP Address
! Link to Main Campus Router
! -------------------------------------------------------
interface Serial0/2/0
 ip address 20.0.0.2 255.255.255.252
 clock rate 64000

! -------------------------------------------------------
! GigabitEthernet connects to the email server
! -------------------------------------------------------
interface GigabitEthernet0/0
 ip address 20.0.0.1 255.255.255.252

! -------------------------------------------------------
! RIPv2
! -------------------------------------------------------
router rip
 version 2
 network 20.0.0.0
 no auto-summary

! -------------------------------------------------------
! Static Route back into the university network
! -------------------------------------------------------
ip route 10.10.10.0 255.255.255.252 20.0.0.1

end
